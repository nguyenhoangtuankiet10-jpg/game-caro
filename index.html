2. Thuật Toán C++ Kiểm Tra Thắng/Thua 5 Nước (Win Checker C++)

Đoạn mã C++ được biên dịch trên Server xử lý kiểm tra 5 quân liên tiếp theo 4 hướng (ngang, dọc, 2 chéo):

// C++ Win Check Algorithm for 15x15 Gomoku/Caro
#include <iostream>
#include <vector>

const int BOARD_SIZE = 15;

bool checkWinCPlusPlus(const std::vector<std::vector<int>>& board, int row, int col, int player) {
    // 4 Hướng: {Ngang, Dọc, Chéo Chính, Chéo Phụ}
    int dx[] = {1, 0, 1, 1};
    int dy[] = {0, 1, 1, -1};

    for (int dir = 0; dir < 4; ++dir) {
        int count = 1;

        // Tiến về phía trước theo hướng dir
        for (int step = 1; step < 5; ++step) {
            int r = row + dy[dir] * step;
            int c = col + dx[dir] * step;
            if (r >= 0 && r < BOARD_SIZE && c >= 0 && c < BOARD_SIZE && board[r][c] == player) count++;
            else break;
        }

        // Lùi về phía sau theo hướng ngược lại
        for (int step = 1; step < 5; ++step) {
            int r = row - dy[dir] * step;
            int c = col - dx[dir] * step;
            if (r >= 0 && r < BOARD_SIZE && c >= 0 && c < BOARD_SIZE && board[r][c] == player) count++;
            else break;
        }

        if (count >= 5) return true; // Thắng!
    }
    return false;
}

3. Cấu Trúc Cơ Sở Dữ Liệu (MySQL / SQL Server DDL Schema)

-- Bảng Người Dùng (Users)
CREATE TABLE Users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    total_wins INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Bảng Lịch Sử Trận Đấu (Matches)
CREATE TABLE Matches (
    match_id INT AUTO_INCREMENT PRIMARY KEY,
    player_x_id INT,
    player_o_id INT,
    winner_id INT,
    total_moves INT,
    played_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (player_x_id) REFERENCES Users(user_id),
    FOREIGN KEY (player_o_id) REFERENCES Users(user_id)
);

-- Bảng Chi Tiết Nước Đi (MoveHistory)
CREATE TABLE MoveHistory (
    move_id INT AUTO_INCREMENT PRIMARY KEY,
    match_id INT,
    player_id INT,
    row_index INT NOT NULL,
    col_index INT NOT NULL,
    step_order INT NOT NULL,
    FOREIGN KEY (match_id) REFERENCES Matches(match_id)
);
